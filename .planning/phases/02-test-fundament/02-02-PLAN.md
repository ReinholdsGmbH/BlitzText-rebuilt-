From BlitzText/Hotkeys/HotkeyManager.swift (after Wave-0 refactor):
```swift
extension HotkeyManager {
    struct FlagsInput: Equatable {
        var fn: Bool
        var shift: Bool
        var control: Bool
        var option: Bool
    }

    enum Outcome: Equatable {
        case none
        case start(Mode)
        case stop(Mode)
        case toggle(Mode)
    }

    internal func decideTransition(
        flags: FlagsInput,
        triggerMode: TriggerMode,
        now: Date
    ) -> [Outcome]
}
```
