<objective>
Extrahiere `installEventTap()` aus `HotkeyManager.start()`, ergänze einen 1-Hz-DispatchSource-Watcher auf `AXIsProcessTrusted()` und installiere den Tap automatisch, sobald die Accessibility-Permission erteilt wurde — ohne App-Relaunch. Update den Permission-Hilfetext in SettingsView entsprechend.

Purpose: Schließt STAB-03. Aktuell ist der Permission-Flow eine User-Falle: nach Klick auf "Aktivieren" passiert sichtbar nichts, bis der User die App manuell beendet und neu öffnet (siehe SettingsView.swift Zeile 209-210 und HotkeyManager.swift Zeile 30-35: bei `trusted = false` wird einfach returnt, der Tap wird nie nachträglich aufgebaut).
Output: Auto-Restart des Taps + Doku-Konsistenz.
</objective>

Neue private Methode (Body identisch zum aktuellen Inhalt von Zeilen 36–67, nur in eigene Methode verschoben):
```swift
private func installEventTap() {
    guard eventTap == nil else { return }   // idempotent — kein doppelter Tap
    let mask = (1 << CGEventType.flagsChanged.rawValue) | (1 << CGEventType.keyDown.rawValue)
    let unmanagedSelf = Unmanaged.passUnretained(self)

    let callback: CGEventTapCallBack = { _, type, event, refcon in
        guard let refcon else { return Unmanaged.passUnretained(event) }
        let manager = Unmanaged<HotkeyManager>.fromOpaque(refcon).takeUnretainedValue()
        if type == .flagsChanged {
            manager.handleFlagsChanged(event: event)
        } else if type == .tapDisabledByTimeout || type == .tapDisabledByUserInput {
            Log.write("Tap disabled by system (\(type.rawValue)) — re-enable")
            if let tap = manager.eventTap { CGEvent.tapEnable(tap: tap, enable: true) }
        }
        return Unmanaged.passUnretained(event)
    }

    guard let tap = CGEvent.tapCreate(
        tap: .cgSessionEventTap,
        place: .headInsertEventTap,
        options: .listenOnly,
        eventsOfInterest: CGEventMask(mask),
        callback: callback,
        userInfo: unmanagedSelf.toOpaque()
    ) else {
        Log.write("CGEvent.tapCreate FAILED — Accessibility evtl. doch nicht erteilt.")
        return
    }
    let src = CFMachPortCreateRunLoopSource(kCFAllocatorDefault, tap, 0)
    CFRunLoopAddSource(CFRunLoopGetCurrent(), src, .commonModes)
    CGEvent.tapEnable(tap: tap, enable: true)
    self.eventTap = tap
    self.runLoopSource = src
    Log.write("CGEventTap aktiv — fn-Modifier wird beobachtet.")
}
```

3. Neuer Permission-Watcher (privat). 1 Hz, prompt-frei (`AXIsProcessTrusted()` ohne Options-Dictionary):
```swift
private func startPermissionWatcher() {
    guard permissionTimer == nil else { return }
    let timer = DispatchSource.makeTimerSource(queue: .main)
    timer.schedule(deadline: .now() + 1.0, repeating: 1.0)
    timer.setEventHandler { [weak self] in
        guard let self else { return }
        if AXIsProcessTrusted() {
            Log.write("HotkeyManager: Accessibility erteilt — Tap wird gestartet.")
            self.permissionTimer?.cancel()
            self.permissionTimer = nil
            self.installEventTap()
        }
    }
    timer.resume()
    permissionTimer = timer
}
```

4. Update `stop()` (Zeile 70–79): Watcher mit teardown:
```swift
func stop() {
    permissionTimer?.cancel()
    permissionTimer = nil
    if let tap = eventTap {
        CGEvent.tapEnable(tap: tap, enable: false)
    }
    if let src = runLoopSource {
        CFRunLoopRemoveSource(CFRunLoopGetCurrent(), src, .commonModes)
    }
    eventTap = nil
    runLoopSource = nil
}
```
